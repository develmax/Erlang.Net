3309: void erts_schedule_dist_command(Port *prt, DistEntry *dist_entry)
3310: {
3311:     DistEntry *dep;
3312:     Eterm id;
3313: 
3314:     if (prt) {
3315: 	ERTS_LC_ASSERT(erts_lc_is_port_locked(prt));
3316: 	ASSERT((erts_atomic32_read_nob(&prt->state)
3317: 		& ERTS_PORT_SFLGS_DEAD) == 0);
3318: 
3319:         dep = (DistEntry*) erts_prtsd_get(prt, ERTS_PRTSD_DIST_ENTRY);
3320:         ERTS_ASSUME(dep);
3321: 	id = prt->common.id;
3322:     }
3323:     else {
3324: 	ERTS_ASSUME(dist_entry);
3325: 	ERTS_LC_ASSERT(erts_lc_rwmtx_is_rlocked(&dist_entry->rwmtx)
3326: 			   || erts_lc_rwmtx_is_rwlocked(&dist_entry->rwmtx));
3327: 	ASSERT(is_internal_port(dist_entry->cid));
3328: 
3329:  	dep = dist_entry;
3330: 	id = dep->cid;

3616:                 erts_aint32_t qflgs;
3617: 		ErtsProcList *plp = NULL;
3618:                 Eterm notify_proc = NIL;
3619:                 Sint obsz;
3620:                 int fragments, empty_fill;
3621: 
3622:                 /* Calculate how many fragments to send. This depends on
3623:                    the available space in the distr queue and the amount
3624:                    of remaining reductions. */
3625:                 for (fragments = 0, obsz = 0;
3626:                      (fragments < ctx->fragments
3627:                       && ((ctx->reds > 0
3628:                            && (ctx->ignore_busy
3629:                                || (qsize + obsz < erts_dist_buf_busy_limit)))
3630:                           || ctx->no_trap
3631:                           || ctx->no_suspend));
3632:                      fragments++) {
3633: #ifdef DEBUG
3634:                     int reds = 100;
3635: #else
3636:                     int reds = 10;
3637: #endif
3638:                     if (!ctx->no_trap && !ctx->no_suspend)
3639:                         ctx->reds -= reds;
3640:                     obsz += size_obuf(&ctx->obuf[fragments]);
3641:                 }
3642: 
3643:                 ASSERT(fragments == ctx->fragments ||
3644:                        (!ctx->no_trap && !ctx->no_suspend));
3645: 
3646: 		erts_mtx_lock(&dep->qlock);
3647:                 update_qsizes(dep, &empty_fill, &qsize, obsz,
3648:                               ctx->ignore_busy ? obsz : 0);
3649:                 qflgs = de_qflags_read(dep);
3650: 		if (!(qflgs & ERTS_DE_QFLG_BUSY)
3651:                     && qsize >= erts_dist_buf_busy_limit) {
3652: 		    qflgs = de_qflags_read_set(dep, ERTS_DE_QFLG_BUSY);
3653:                     qflgs |= ERTS_DE_QFLG_BUSY;
3654:                 }
3655:                 if (empty_fill && is_internal_pid(dep->cid)) {
3656:                     erts_aint32_t notify;
3657:                     notify = erts_atomic32_xchg_mb(&dep->notify,
3658:                                                    (erts_aint32_t) 0);
3659:                     if (notify) {
3660:                         /*
3661:                          * Previously empty queue and notification
3662:                          * requested...
3663:                          */
3664:                         notify_proc = dep->cid;
3665:                         ASSERT(is_internal_pid(notify_proc));
3666:                     }
3667:                 }
3668: 
3669:                 ASSERT(fragments < 2
3670:                        || (get_int64(&((char*)ctx->obuf->eiov->iov[1].iov_base)[10])
3671:                            == ctx->fragments));
3672: 
3673:                 if (fragments) {
3674:                     ctx->obuf[fragments-1].next = NULL;
3675:                     if (dep->out_queue.last)
3676:                         dep->out_queue.last->next = ctx->obuf;
3677:                     else
3678:                         dep->out_queue.first = ctx->obuf;
3679:                     dep->out_queue.last = &ctx->obuf[fragments-1];
3680: 
3681:                     ctx->fragments -= fragments;
3682:                     ctx->obuf = &ctx->obuf[fragments];
3683:                 }
3684: 
3685: 		if ((qflgs & ERTS_DE_QFLG_BUSY)
3686:                     && !ctx->ignore_busy
3687:                     && !ctx->no_suspend) {
3688: 
3689:                     erts_mtx_unlock(&dep->qlock);
3690: 
3691:                     plp = erts_proclist_create(ctx->c_p);
3692: 
3693:                     erts_suspend(ctx->c_p, ERTS_PROC_LOCK_MAIN, NULL);
3694:                     suspended = 1;
3695: 
3696:                     erts_mtx_lock(&dep->qlock);
3697: 
3698:                     qflgs = de_qflags_read(dep);
3699:                     if (qflgs & ERTS_DE_QFLG_BUSY) {
3700:                         /* Enqueue suspended process on dist entry */
3701:                         ASSERT(plp);
3702:                         erts_proclist_store_last(&dep->suspended, plp);
3703:                     }
3704:                     else {
3705:                         resume = 1; /* was busy, but isn't now */
3706:     #ifdef USE_VM_PROBES
3707:                         if (resume && DTRACE_ENABLED(dist_port_not_busy)) {

3709:                             DTRACE_CHARBUF(remote_str, 64);

3711:                             erts_snprintf(port_str, sizeof(DTRACE_CHARBUF_NAME(port_str)),

3714:                                           "%T", dep->sysname);
3715:                             DTRACE3(dist_port_not_busy, erts_this_node_sysname,
3716:                                     port_str, remote_str);
3717:                         }
3718:     #endif
3719:                     }
3720: 		}
3721: 
3722: 		erts_mtx_unlock(&dep->qlock);
3723: 		if (dep->state != ERTS_DE_STATE_PENDING) {
3724:                     if (is_internal_port(dep->cid))
3725:                         erts_schedule_dist_command(NULL, dep);
3726:                 }
3727:                 else {
3728:                     notify_proc = NIL;
3729:                 }
3730: 		erts_de_runlock(dep);
3731:                 if (is_internal_pid(notify_proc))
3732:                     notify_dist_data(ctx->c_p, notify_proc);
3733: 
3734: 		if (resume) {
3735: 		    erts_resume(ctx->c_p, ERTS_PROC_LOCK_MAIN);
3736: 		    erts_proclist_destroy(plp);
3737: 		    /*
3738: 		     * Note that the calling process still have to yield as if it
3739: 		     * suspended. If not, the calling process could later be
3740: 		     * erroneously scheduled when it shouldn't be.
3741: 		     */
3742: 		}
3743:                 /* More fragments left to be sent, yield and re-schedule */
3744:                 if (ctx->fragments) {
3745:                     ctx->c_p->flags |= F_FRAGMENTED_SEND;
3746:                     retval = ERTS_DSIG_SEND_CONTINUE;
3747:                     if (!resume && erts_system_monitor_busy_dist_port_cnt)
3748:                         monitor_busy_dist_port(ctx->c_p, cid);
3749:                     goto done;
3750:                 }
3751: 	    }
3752: 
3753:             if (ctx->c_p) ctx->c_p->flags &= ~F_FRAGMENTED_SEND;
3754: 	    ctx->obuf = NULL;
3755: 
3756: 	    if (suspended) {
3757:     #ifdef USE_VM_PROBES

3760: 		    DTRACE_CHARBUF(remote_str, 64);
3761: 		    DTRACE_CHARBUF(pid_str, 16);
3762: 
3763: 		    erts_snprintf(port_str, sizeof(DTRACE_CHARBUF_NAME(port_str)), "%T", cid);
3764: 		    erts_snprintf(remote_str, sizeof(DTRACE_CHARBUF_NAME(remote_str)),
3765: 				  "%T", dep->sysname);
3766: 		    erts_snprintf(pid_str, sizeof(DTRACE_CHARBUF_NAME(pid_str)),
3767: 				  "%T", ctx->c_p->common.id);

3769: 			    port_str, remote_str, pid_str);

3771:     #endif
3772:                 if (!resume && erts_system_monitor_busy_dist_port_cnt) {
3773:                     monitor_busy_dist_port(ctx->c_p, cid);
3774:                 }

3776: 	    } else {

3778: 	    }

3780: 	}
3781: 	default:

3783: 	}
3784:     }
3785: 
3786: done:
3787:     if (ctx->msg && ctx->c_p) {

3789:     }
3790:     return retval;
3791: }
