5714: {
5715:     erts_de_rwlock(dep);
5716:     if (dep->state != ERTS_DE_STATE_IDLE) {
5717:         erts_de_rwunlock(dep);
5718:     }
5719:     else {
5720:         Process* net_kernel;
5721:         ErtsProcLocks nk_locks = ERTS_PROC_LOCK_MSGQ;
5722:         Eterm *hp;
5723:         ErlOffHeap *ohp;
5724:         ErtsMessage *mp;
5725:         Eterm msg, dhandle;
5726:         Uint32 conn_id;
5727: 
5728:         erts_set_dist_entry_pending(dep);
5729:         conn_id = dep->connection_id;
5730:         erts_de_rwunlock(dep);
5731: 
5732:         net_kernel = erts_whereis_process(proc, proc_locks,
5733:                                           am_net_kernel, nk_locks, 0);
5734:         if (!net_kernel) {
5735:             abort_pending_connection(dep, conn_id, NULL);
5736:             return 0;
5737:         }
5738: 
5739:         if (proc == net_kernel)
5740:             nk_locks |= ERTS_PROC_LOCK_MAIN;

5743:          * Send {auto_connect, Node, DHandle} to net_kernel
5744:          */

5746:                                      4 + ERTS_DHANDLE_SIZE,
5747:                                      &hp, &ohp);

5749:         msg = TUPLE3(hp, am_auto_connect, dep->sysname, dhandle);
5750:         ERL_MESSAGE_TOKEN(mp) = am_undefined;
5751:         erts_queue_proc_message(proc, net_kernel, nk_locks, mp, msg);
5752: 
5753:         if (proc == net_kernel)
5754:             nk_locks &= ~ERTS_PROC_LOCK_MAIN;
5755: 
5756:         erts_proc_unlock(net_kernel, nk_locks);
5757:     }

5759:     return 1;
