using System.Reflection;
using System.Runtime.Loader;

string tests = Path.GetFullPath(args[0]);
string directory = Path.GetDirectoryName(tests)!;
AssemblyLoadContext.Default.Resolving += (context, name) =>
{
    string path = Path.Combine(directory, name.Name + ".dll");
    return File.Exists(path) ? context.LoadFromAssemblyPath(path) : null;
};
Assembly assembly = AssemblyLoadContext.Default.LoadFromAssemblyPath(tests);
MethodInfo check = assembly.GetType("Erlang.Tests.BeamValueMapDefaultChecks", throwOnError: true)!
    .GetMethod("CheckComputedKey", BindingFlags.Static | BindingFlags.NonPublic)!;
for (int i = 1; i < args.Length; i++)
{
    try
    {
        await (Task)check.Invoke(null, [File.ReadAllBytes(args[i])])!;
        Console.WriteLine($"{i} Passed");
        if (i == 1) return 1;
    }
    catch (InvalidOperationException exception) when (i == 1)
    {
        Console.WriteLine($"{i} Failed: {exception.Message}");
        if (!exception.Message.Contains("changed eager operand order")) return 1;
    }
}
return 0;
